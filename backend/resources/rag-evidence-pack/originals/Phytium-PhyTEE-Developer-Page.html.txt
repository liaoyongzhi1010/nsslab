<p><span style="font-size:12pt"><span style="font-family:&quot;Times New Roman&quot;"><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'"><span style="font-family:宋体">以飞腾安全处理器为核心，通过芯片级硬安全组件构建底层根基：</span></span></span></span></span></p>

<p><span style="font-size:12pt"><span style="font-family:&quot;Times New Roman&quot;"><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'">1.PhyTCM/PhyTPCM<span style="font-family:宋体">：提供硬件可信根，保障系统启动与运行的可信基础</span></span></span></span></span></p>

<p><span style="font-size:12pt"><span style="font-family:&quot;Times New Roman&quot;"><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'">2.PhyCrypto<span style="font-family:宋体">：承载</span></span></span><span style="font-size:12.0000pt"><span style="font-family:宋体"><span style="font-family:宋体">密码</span></span></span><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'"><span style="font-family:宋体">算法加速能力，为密码应用提供硬件级支撑</span></span></span></span></span></p>

<p><span style="font-size:12pt"><span style="font-family:&quot;Times New Roman&quot;"><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'">3.PhyTEE/PhyCCA<span style="font-family:宋体">：构建可信执行环境与机密计算架构，实现数据隔离与安全运行</span></span></span></span></span></p>

<p><span style="font-size:12pt"><span style="font-family:&quot;Times New Roman&quot;"><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'"><span style="font-family:宋体">二、核心安全生态</span></span></span></span></span></p>

<p><span style="font-size:12pt"><span style="font-family:&quot;Times New Roman&quot;"><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'">1.<span style="font-family:宋体">可信生态：以硬件可信根为起点，实现系统全生命周期可信度量，从源头筑牢系统安全</span></span></span></span></span></p>

<p><span style="font-size:12pt"><span style="font-family:&quot;Times New Roman&quot;"><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'">2.<span style="font-family:宋体">密码生态：依托硬件加速，搭建全栈密码框架，适配主流标准接口，满足行业合规要求</span></span></span></span></span></p>

<p><span style="font-size:12pt"><span style="font-family:&quot;Times New Roman&quot;"><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'">3.<span style="font-family:宋体">机密计算生态：构建高等级隔离环境，实现数据 </span><span style="font-family:Times New Roman,Times,serif">&ldquo;</span><span style="font-family:宋体">可用不可见</span><span style="font-family:Times New Roman,Times,serif">&rdquo;</span><span style="font-family:宋体">，破解数据安全与流通的核心矛盾</span></span></span></span></span></p>

<p><span style="font-size:12pt"><span style="font-family:&quot;Times New Roman&quot;"><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'"><span style="font-family:宋体">三、核心优势与价值</span></span></span></span></span></p>

<p><span style="font-size:12pt"><span style="font-family:&quot;Times New Roman&quot;"><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'">1.<span style="font-family:宋体">自主可控：全栈自主研发，摆脱外部依赖，保障核心技术安全</span></span></span></span></span></p>

<p><span style="font-size:12pt"><span style="font-family:&quot;Times New Roman&quot;"><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'">2.<span style="font-family:宋体">全栈覆盖：整合三大安全能力，覆盖信息安全全生命周期需求</span></span></span></span></span></p>

<p><span style="font-size:12.0000pt"><span style="font-family:'Times New Roman'">3.<span style="font-family:宋体">场景适配：深度兼容国产生态，降低迁移成本，高效赋能金融、政务、能源等关键行业</span></span></span></p>

        // 获取左侧图片宽度，动态设置 --layout-content-width
        function updateLayoutWidth() {
            const container = document.querySelector('.sec-body');
            const leftImg = document.getElementById('architecture-left-img');
            if (!leftImg || !leftImg.complete) return;
            // 获取左侧图片实际显示宽度
            const height = leftImg.offsetHeight;
            // 设置 CSS 变量
            container.style.setProperty('--layout-content-width', height + 'px');
        }

        // 图片加载完成时计算一次
        if (document.getElementById('architecture-left-img')) {
            document.getElementById('architecture-left-img').addEventListener('load', updateLayoutWidth);
        }

        // 页面加载完成尝试计算一次
        window.addEventListener('load', updateLayoutWidth);

        // 窗口大小变化时重新计算
        window.addEventListener('resize', updateLayoutWidth);
